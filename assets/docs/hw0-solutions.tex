\problemheader{Problem 1: Expected Prediction Error \& Bias-Variance in $k$-NN}{[25 points]}

\textbf{Problem Statement:}
Consider the classical nonparametric regression setting. Suppose we have a training dataset $\mathcal{D} = \{(x_i, y_i)\}_{i=1}^N$ where the scalar targets are generated by the model $y = f(x) + \epsilon$, with conditional zero mean $\mathbb{E}[\epsilon \mid x] = 0$ and homoscedastic variance $\operatorname{Var}(\epsilon \mid x) = \sigma^2$. The additive noise $\epsilon$ is mutually independent across observations and independent of the input vector $x \in \mathbb{R}^p$.

For a fixed query test point $x_0 \in \mathbb{R}^p$:
\begin{enumerate}[label=(\alph*), itemsep=3pt, leftmargin=18pt]
  \item Show that under squared-error loss, the conditional Expected Prediction Error of the $1$-NN estimator $\hat{f}_1(x_0) = y_{(1)}$ decomposes into irreducible noise, squared bias, and estimator variance.
  \item Generalize the decomposition to the $k$-NN estimator $\hat{f}_k(x_0) = \frac{1}{k}\sum_{\ell=1}^k y_{(\ell)}$.
  \item Analyze the trade-off as $k$ varies from $1$ to $N$, and explain why bias fails to vanish in high dimensions ($p \gg 1$).
\end{enumerate}

\vspace{8pt}
\noindent{\large\bfseries\textcolor{utblue}{Solution:}}
\vspace{4pt}

\textbf{Part (a): 1-Nearest-Neighbor Decomposition [9 points]} \\
Let $(x_0, y_0)$ be an independent out-of-sample query observation with response $y_0 = f(x_0) + \epsilon_0$, where $\epsilon_0$ is independent of the training data $\mathcal{D}$ and satisfies $\mathbb{E}[\epsilon_0] = 0$ and $\operatorname{Var}(\epsilon_0) = \sigma^2$.

The $1$-Nearest-Neighbor estimator evaluated at $x_0$ is:
\[
  \hat{f}_1(x_0) = y_{(1)} = f(x_{(1)}) + \epsilon_{(1)},
\]
where $x_{(1)} \in \{x_1, \dots, x_N\}$ denotes the training input closest to $x_0$ in Euclidean distance, and $\epsilon_{(1)}$ is its corresponding training observation noise.

Subtracting the estimator from the test target:
\[
  y_0 - \hat{f}_1(x_0) = \left( f(x_0) + \epsilon_0 \right) - \left( f(x_{(1)}) + \epsilon_{(1)} \right) = \left[ f(x_0) - f(x_{(1)}) \right] + \epsilon_0 - \epsilon_{(1)}.
\]
Now we square this error and take the conditional expectation with respect to the test noise $\epsilon_0$ and the training targets conditioned on the fixed sample locations $\{x_i\}_{i=1}^N$ and query point $x_0$:
\begin{align*}
  \operatorname{EPE}(x_0) &\equiv \mathbb{E}\left[ (y_0 - \hat{f}_1(x_0))^2 \;\middle|\; x_0, \{x_i\}_{i=1}^N \right] \\
  &= \mathbb{E}\left[ \left( [f(x_0) - f(x_{(1)})] + (\epsilon_0 - \epsilon_{(1)}) \right)^2 \;\middle|\; x_0, \{x_i\}_{i=1}^N \right] \\
  &= \left[ f(x_0) - f(x_{(1)}) \right]^2 + \mathbb{E}[\epsilon_0^2] + \mathbb{E}[\epsilon_{(1)}^2] \\
  &\quad + 2\left[ f(x_0) - f(x_{(1)}) \right] \underbrace{\mathbb{E}[\epsilon_0]}_{=0} - 2\left[ f(x_0) - f(x_{(1)}) \right] \underbrace{\mathbb{E}[\epsilon_{(1)}]}_{=0} - 2\underbrace{\mathbb{E}[\epsilon_0 \epsilon_{(1)}]}_{=0}.
\end{align*}
Because $\epsilon_0$ and $\epsilon_{(1)}$ are independent zero-mean random variables with $\mathbb{E}[\epsilon_0^2] = \sigma^2$ and $\mathbb{E}[\epsilon_{(1)}^2] = \sigma^2$, all interaction cross-terms vanish identically:
\[
  \operatorname{EPE}(x_0) = \underbrace{\sigma^2}_{\text{Irreducible Error } \operatorname{Var}(\epsilon_0)} + \underbrace{\left[ f(x_0) - f(x_{(1)}) \right]^2}_{\text{Squared Bias } [\operatorname{Bias}(\hat{f}_1(x_0))]^2} + \underbrace{\sigma^2}_{\text{Sampling Variance } \operatorname{Var}(\hat{f}_1(x_0))}. \quad \blacksquare
\]
The three fundamental components are:
\begin{itemize}[itemsep=2pt, leftmargin=18pt]
  \item \textbf{Irreducible Noise ($\sigma^2$):} The intrinsic variation in the target $y_0$ that no model, regardless of complexity, can predict.
  \item \textbf{Squared Model Bias ($[f(x_0) - f(x_{(1)})]^2$):} The systematic geometric approximation error arising from using the true function value at the closest training sample $x_{(1)}$ instead of the target point $x_0$.
  \item \textbf{Estimator Variance ($\sigma^2$):} The sampling variability of the estimator induced by the observation noise $\epsilon_{(1)}$ at the selected neighbor.
\end{itemize}

\vspace{4pt}
\textbf{Part (b): Generalization to $k$-Nearest Neighbors [9 points]} \\
For the $k$-NN averaging estimator:
\[
  \hat{f}_k(x_0) = \frac{1}{k} \sum_{\ell=1}^k y_{(\ell)} = \frac{1}{k}\sum_{\ell=1}^k f(x_{(\ell)}) + \frac{1}{k}\sum_{\ell=1}^k \epsilon_{(\ell)}.
\]
Conditioning on the fixed positions $\{x_i\}_{i=1}^N$:
\begin{enumerate}[label=\arabic*., leftmargin=18pt, itemsep=3pt]
  \item \textbf{Conditional Expectation:}
  \[
    \mathbb{E}\left[\hat{f}_k(x_0) \;\middle|\; \{x_i\}\right] = \frac{1}{k}\sum_{\ell=1}^k f(x_{(\ell)}) + \frac{1}{k}\sum_{\ell=1}^k \underbrace{\mathbb{E}[\epsilon_{(\ell)}]}_{=0} = \frac{1}{k}\sum_{\ell=1}^k f(x_{(\ell)}).
  \]
  \item \textbf{Squared Bias:}
  \[
    \operatorname{Bias}\left(\hat{f}_k(x_0)\right) = \mathbb{E}\left[\hat{f}_k(x_0)\right] - f(x_0) = \left( \frac{1}{k}\sum_{\ell=1}^k f(x_{(\ell)}) \right) - f(x_0).
  \]
  Squaring this quantity yields $\left[ f(x_0) - \frac{1}{k}\sum_{\ell=1}^k f(x_{(\ell)}) \right]^2$.
  \item \textbf{Estimator Variance:} Since the noise terms $\epsilon_{(\ell)}$ are mutually independent with common variance $\sigma^2$:
  \[
    \operatorname{Var}\left(\hat{f}_k(x_0) \;\middle|\; \{x_i\}\right) = \operatorname{Var}\left( \frac{1}{k}\sum_{\ell=1}^k \epsilon_{(\ell)} \right) = \frac{1}{k^2} \sum_{\ell=1}^k \operatorname{Var}(\epsilon_{(\ell)}) = \frac{1}{k^2}(k \sigma^2) = \frac{\sigma^2}{k}.
  \]
\end{enumerate}
Summing the irreducible noise, squared bias, and variance gives the exact analytical formula:
\[
  \operatorname{EPE}(x_0; k) = \sigma^2 + \left[ f(x_0) - \frac{1}{k}\sum_{\ell=1}^k f(x_{(\ell)}) \right]^2 + \frac{\sigma^2}{k}. \quad \blacksquare
\]

\vspace{4pt}
\textbf{Part (c): Complexity Trade-Off and the Curse of Dimensionality [7 points]} \\
\begin{itemize}[itemsep=3pt, leftmargin=18pt]
  \item \textbf{Extreme $k = 1$ (High Model Complexity / Overfitting):}
  Variance achieves its highest value $\operatorname{Var} = \sigma^2$. The model fits local training noise aggressively. However, bias is minimized because only the closest available training point $x_{(1)}$ is queried.
  \item \textbf{Extreme $k = N$ (Low Model Complexity / Underfitting):}
  Variance is suppressed to its minimum $\operatorname{Var} = \frac{\sigma^2}{N} \approx 0$. However, the estimator reduces to predicting the global training sample average $\bar{y}$, making the bias $\left[ f(x_0) - \frac{1}{N}\sum_{i=1}^N f(x_i) \right]^2$ very large whenever $f$ is non-constant.
  \item \textbf{The Curse of Dimensionality in High Dimensions ($p \gg 1$):}
  In $\mathbb{R}^p$, the volume of a hypercube of side length $d \in (0, 1)$ scales as $d^p$. To capture a fixed fraction $r$ of the data points, the neighborhood diameter must scale as $d \approx r^{1/p}$. As $p$ grows large, $r^{1/p} \to 1$ even for small fractions $r$ (e.g., for $r = 0.01$ and $p = 20$, $d = 0.01^{1/20} \approx 0.794$). Consequently, all training points become equidistant and lie on the boundary of the space. Even the closest neighbor $x_{(1)}$ is geometrically far from $x_0$, so $|f(x_0) - f(x_{(1)})| \not\to 0$. The bias term fails to vanish, causing nonparametric nearest-neighbor models to break down without severe dimension reduction or structural regularization. $\blacksquare$
\end{itemize}

\problemheader{Problem 2: Ridge Regression as an Augmented OLS System}{[25 points]}

\textbf{Problem Statement:}
In high-dimensional and collinear data regimes, Ordinary Least Squares can suffer from severe variance inflation or non-invertibility of $X^T X$. Ridge regression stabilizes parameter estimates by penalizing the $L_2$-norm of the regression coefficients:
\[
  \min_{\beta \in \mathbb{R}^p} \left\{ \|y - X\beta\|_2^2 + \lambda \|\beta\|_2^2 \right\}, \quad (\lambda > 0).
\]
Define the augmented feature matrix $X^* = \begin{bmatrix} X \\ \sqrt{\lambda} I_p \end{bmatrix}$ and augmented response $y^* = \begin{bmatrix} y \\ 0_p \end{bmatrix}$.
\begin{enumerate}[label=(\alph*), itemsep=3pt, leftmargin=18pt]
  \item Define the augmented data system $(X^*, y^*)$.
  \item Show algebraically that the augmented OLS problem $\min_\beta \|y^* - X^* \beta\|_2^2$ is identically equal to the Ridge regression objective.
  \item Show that the closed-form augmented OLS solution reproduces $\hat{\beta}_{\text{ridge}} = (X^T X + \lambda I_p)^{-1} X^T y$.
  \item Prove that $X^{*T} X^*$ is strictly positive definite for any $\lambda > 0$, guaranteeing invertibility even when $N < p$ or when columns of $X$ are collinear.
\end{enumerate}

\vspace{8pt}
\noindent{\large\bfseries\textcolor{utblue}{Solution:}}
\vspace{4pt}

\textbf{Part (a): Augmented Data System Definition [5 points]} \\
We define the vertically stacked augmented feature matrix $X^* \in \mathbb{R}^{(N+p) \times p}$ and response vector $y^* \in \mathbb{R}^{N+p}$ as:
\[
  X^* = \begin{bmatrix} X \\ \sqrt{\lambda} I_p \end{bmatrix}, \quad y^* = \begin{bmatrix} y \\ 0_p \end{bmatrix},
\]
where $I_p$ denotes the $p \times p$ identity matrix and $0_p$ is a column vector of $p$ zeros. The first $N$ rows correspond to the original empirical data, while the bottom $p$ rows represent synthetic virtual observations penalizing deviation from zero.

\vspace{4pt}
\textbf{Part (b): Equivalence of the Objectives [6 points]} \\
Compute the augmented residual vector by matrix-vector block arithmetic:
\[
  y^* - X^* \beta = \begin{bmatrix} y \\ 0_p \end{bmatrix} - \begin{bmatrix} X \\ \sqrt{\lambda} I_p \end{bmatrix} \beta = \begin{bmatrix} y - X\beta \\ 0_p - \sqrt{\lambda} \beta \end{bmatrix} = \begin{bmatrix} y - X\beta \\ -\sqrt{\lambda} \beta \end{bmatrix}.
\]
The squared Euclidean norm of a partitioned vector is the sum of the squared Euclidean norms of its subvectors:
\begin{align*}
  \|y^* - X^* \beta\|_2^2 &= \left\| \begin{bmatrix} y - X\beta \\ -\sqrt{\lambda} \beta \end{bmatrix} \right\|_2^2 = \|y - X\beta\|_2^2 + \|-\sqrt{\lambda} \beta\|_2^2 \\
  &= \|y - X\beta\|_2^2 + (\sqrt{\lambda})^2 \|\beta\|_2^2 \\
  &= \|y - X\beta\|_2^2 + \lambda \|\beta\|_2^2.
\end{align*}
This proves that the standard ordinary least squares objective on the augmented dataset $(X^*, y^*)$ is identically equal to the penalized Ridge regression objective for any candidate $\beta \in \mathbb{R}^p$. $\blacksquare$

\vspace{4pt}
\textbf{Part (c): Equivalence of Closed-Form Estimators [7 points]} \\
Compute the Gram matrix of the augmented feature matrix $X^*$:
\[
  X^{*T} X^* = \begin{bmatrix} X^T & \sqrt{\lambda} I_p \end{bmatrix} \begin{bmatrix} X \\ \sqrt{\lambda} I_p \end{bmatrix} = X^T X + (\sqrt{\lambda} I_p)(\sqrt{\lambda} I_p) = X^T X + \lambda I_p.
\]
Next, compute the cross-product vector between $X^*$ and $y^*$:
\[
  X^{*T} y^* = \begin{bmatrix} X^T & \sqrt{\lambda} I_p \end{bmatrix} \begin{bmatrix} y \\ 0_p \end{bmatrix} = X^T y + \sqrt{\lambda} I_p (0_p) = X^T y.
\]
The normal equations for the unconstrained augmented OLS problem state that $\hat{\beta}^*$ must satisfy:
\[
  (X^{*T} X^*) \hat{\beta}^* = X^{*T} y^* \quad \Longleftrightarrow \quad (X^T X + \lambda I_p) \hat{\beta}^* = X^T y.
\]
Assuming invertibility (proven in Part (d)), solving for $\hat{\beta}^*$ gives:
\[
  \hat{\beta}^* = (X^{*T} X^*)^{-1} X^{*T} y^* = (X^T X + \lambda I_p)^{-1} X^T y = \hat{\beta}_{\text{ridge}}. \quad \blacksquare
\]

\vspace{4pt}
\textbf{Part (d): Strict Positive Definiteness and Numerical Invertibility [7 points]} \\
Let $v \in \mathbb{R}^p \setminus \{0_p\}$ be any non-zero vector. Evaluate the quadratic form associated with $X^{*T} X^*$:
\[
  v^T (X^{*T} X^*) v = v^T (X^T X + \lambda I_p) v = v^T X^T X v + \lambda v^T v = \|X v\|_2^2 + \lambda \|v\|_2^2.
\]
\begin{itemize}[itemsep=2pt, leftmargin=18pt]
  \item Since the Euclidean norm is always non-negative, $\|X v\|_2^2 \ge 0$.
  \item Since $v \ne 0_p$, its squared norm is strictly positive: $\|v\|_2^2 > 0$.
  \item Since the regularization parameter $\lambda > 0$, the product $\lambda \|v\|_2^2 > 0$.
\end{itemize}
Therefore, we have the strict inequality:
\[
  v^T (X^{*T} X^*) v = \|X v\|_2^2 + \lambda \|v\|_2^2 \ge 0 + \lambda \|v\|_2^2 > 0 \quad \forall v \ne 0_p.
\]
This proves that $X^{*T} X^* = X^T X + \lambda I_p$ is \textbf{strictly positive definite} for any $\lambda > 0$.

\textbf{Implications for Machine Learning and Ill-Posed Systems:}
\begin{itemize}[itemsep=2pt, leftmargin=18pt]
  \item \textbf{Case $N < p$ (High-Dimensional Regime):} The raw Gram matrix $X^T X$ has rank at most $N < p$, meaning it has at least $p - N$ zero eigenvalues and is completely non-invertible in standard OLS. By adding $\lambda I_p$, all eigenvalues $\mu_j(X^T X)$ are shifted upward by $\lambda$:
  \[
    \mu_j(X^T X + \lambda I_p) = \mu_j(X^T X) + \lambda \ge 0 + \lambda = \lambda > 0.
  \]
  Thus, zero eigenvalues are eliminated, guaranteeing a unique, well-defined solution exists even when $p \gg N$.
  \item \textbf{Multicollinearity \& Numerical Conditioning:} When features are highly correlated, the condition number $\kappa(X^T X) = \frac{\mu_{\max}}{\mu_{\min}} \to \infty$, leading to catastrophic numerical error and massive coefficient variance. Ridge regularization bounds the condition number by $\kappa(X^T X + \lambda I_p) \le \frac{\mu_{\max} + \lambda}{\lambda} < \infty$, stabilizing the system against matrix inversion perturbations. $\blacksquare$
\end{itemize}

\problemheader{Problem 3: Multiclass Softmax Regression \& Log-Likelihood Concavity}{[25 points]}

\textbf{Problem Statement:}
Consider a multiclass classification problem with $K \ge 2$ classes and 1-of-$K$ target encoding $y_i = [y_{i1}, \dots, y_{iK}]^T \in \{0, 1\}^K$ with $\sum_{k=1}^K y_{ik} = 1$. The class posterior probabilities follow the Softmax formulation:
\[
  p_k(x_i; \theta) \equiv P(Y = k \mid x_i) = \frac{\exp(\beta_k^T x_i)}{\sum_{\ell=1}^K \exp(\beta_\ell^T x_i)}, \quad k = 1, \dots, K.
\]
\begin{enumerate}[label=(\alph*), itemsep=3pt, leftmargin=18pt]
  \item Formulate the conditional log-likelihood function $\ell(\theta) = \sum_{i=1}^N \sum_{k=1}^K y_{ik} \log p_k(x_i; \theta)$.
  \item Derive the score gradient $\nabla_{\beta_k} \ell(\theta) = \sum_{i=1}^N (y_{ik} - p_k(x_i; \theta)) x_i$ and interpret it.
  \item Compute the block Hessian $\mathcal{H}_{k\ell} = -\sum_{i=1}^N p_k(x_i)(\delta_{k\ell} - p_\ell(x_i)) x_i x_i^T$.
  \item Prove that $-\mathcal{H} \succeq 0$, establishing global concavity and uniqueness of the maximum likelihood estimate.
\end{enumerate}

\vspace{8pt}
\noindent{\large\bfseries\textcolor{utblue}{Solution:}}
\vspace{4pt}

\textbf{Part (a): Formulation of the Conditional Log-Likelihood [5 points]} \\
For an independent sample $i$, the response vector $y_i$ is governed by a single-trial multinomial distribution with event probabilities $p_1(x_i), \dots, p_K(x_i)$. The probability of observation $(x_i, y_i)$ is:
\[
  P(y_i \mid x_i; \theta) = \prod_{k=1}^K [p_k(x_i; \theta)]^{y_{ik}}.
\]
Taking the product over all $N$ independent observations in $\mathcal{D}$ and computing the natural logarithm yields the total log-likelihood function:
\[
  \ell(\theta) = \ln \prod_{i=1}^N \prod_{k=1}^K [p_k(x_i; \theta)]^{y_{ik}} = \sum_{i=1}^N \sum_{k=1}^K y_{ik} \ln p_k(x_i; \theta). \quad \blacksquare
\]

\vspace{4pt}
\textbf{Part (b): Score Equations and Physical Interpretation [7 points]} \\
Define the linear predictor score $\eta_{ik} = \beta_k^T x_i$. By the chain rule, for any class $m \in \{1, \dots, K\}$:
\[
  \frac{\partial p_m(x_i)}{\partial \eta_{ik}} = \frac{\partial}{\partial \eta_{ik}} \left[ \frac{\exp(\eta_{im})}{\sum_{\ell=1}^K \exp(\eta_{i\ell})} \right].
\]
Using the quotient rule:
\begin{itemize}[itemsep=2pt, leftmargin=18pt]
  \item If $m = k$: $\frac{\partial p_k(x_i)}{\partial \eta_{ik}} = \frac{\exp(\eta_{ik})\sum_\ell \exp(\eta_{i\ell}) - \exp(\eta_{ik})\exp(\eta_{ik})}{\left( \sum_\ell \exp(\eta_{i\ell}) \right)^2} = p_k(x_i)(1 - p_k(x_i))$.
  \item If $m \ne k$: $\frac{\partial p_m(x_i)}{\partial \eta_{ik}} = \frac{0 - \exp(\eta_{im})\exp(\eta_{ik})}{\left( \sum_\ell \exp(\eta_{i\ell}) \right)^2} = -p_m(x_i) p_k(x_i)$.
\end{itemize}
Combining via the Kronecker delta $\delta_{km}$:
\[
  \frac{\partial p_m(x_i)}{\partial \eta_{ik}} = p_m(x_i) \left( \delta_{km} - p_k(x_i) \right).
\]
Now differentiate the complete log-likelihood $\ell(\theta)$ with respect to $\beta_k$:
\begin{align*}
  \frac{\partial \ell(\theta)}{\partial \beta_k} &= \sum_{i=1}^N \sum_{m=1}^K y_{im} \frac{1}{p_m(x_i)} \frac{\partial p_m(x_i)}{\partial \beta_k} \\
  &= \sum_{i=1}^N \sum_{m=1}^K y_{im} \frac{1}{p_m(x_i)} \left[ p_m(x_i)(\delta_{km} - p_k(x_i)) x_i \right] \\
  &= \sum_{i=1}^N \left( \sum_{m=1}^K y_{im} \delta_{km} - p_k(x_i) \sum_{m=1}^K y_{im} \right) x_i.
\end{align*}
Because $\sum_{m=1}^K y_{im} \delta_{km} = y_{ik}$ and $\sum_{m=1}^K y_{im} = 1$ (mutually exclusive encoding):
\[
  \nabla_{\beta_k} \ell(\theta) = \sum_{i=1}^N \left( y_{ik} - p_k(x_i; \theta) \right) x_i. \quad \blacksquare
\]
\textbf{Interpretation:} The gradient vector for class $k$ is an error-weighted superposition of feature inputs:
\begin{itemize}[itemsep=2pt, leftmargin=18pt]
  \item The term $(y_{ik} - p_k(x_i))$ represents the empirical prediction residual.
  \item If sample $i$ truly belongs to class $k$ ($y_{ik} = 1$), the gradient pushes $\beta_k$ along $+x_i$ with magnitude proportional to the underconfidence $(1 - p_k(x_i))$.
  \item If sample $i$ belongs to a different class ($y_{ik} = 0$), the residual is $-p_k(x_i)$, which pushes $\beta_k$ in the direction $-x_i$ to penalize unwarranted overconfidence.
\end{itemize}

\vspace{4pt}
\textbf{Part (c): Hessian Matrix Derivation [6 points]} \\
To compute the second-order partial derivative block $\mathcal{H}_{k\ell} \equiv \frac{\partial^2 \ell(\theta)}{\partial \beta_k \partial \beta_\ell^T} \in \mathbb{R}^{(p+1) \times (p+1)}$:
\[
  \frac{\partial^2 \ell(\theta)}{\partial \beta_k \partial \beta_\ell^T} = \frac{\partial}{\partial \beta_\ell^T} \left[ \sum_{i=1}^N (y_{ik} - p_k(x_i)) x_i \right] = -\sum_{i=1}^N x_i \left( \frac{\partial p_k(x_i)}{\partial \beta_\ell^T} \right).
\]
Applying the chain rule $\frac{\partial p_k(x_i)}{\partial \beta_\ell^T} = \frac{\partial p_k(x_i)}{\partial \eta_{i\ell}} x_i^T = p_k(x_i)(\delta_{k\ell} - p_\ell(x_i)) x_i^T$:
\[
  \mathcal{H}_{k\ell} = -\sum_{i=1}^N p_k(x_i; \theta) \left( \delta_{k\ell} - p_\ell(x_i; \theta) \right) x_i x_i^T. \quad \blacksquare
\]

\vspace{4pt}
\textbf{Part (d): Positive Semi-Definiteness and Global Concavity [7 points]} \\
Let $u = [u_1^T, \dots, u_K^T]^T \in \mathbb{R}^{K(p+1)}$ be an arbitrary test vector, where each $u_k \in \mathbb{R}^{p+1}$.
Examine the quadratic form of the negative Hessian matrix $-\mathcal{H} = -\nabla^2 \ell(\theta)$:
\begin{align*}
  u^T (-\mathcal{H}) u &= \sum_{k=1}^K \sum_{\ell=1}^K u_k^T \left[ \sum_{i=1}^N p_k(x_i) (\delta_{k\ell} - p_\ell(x_i)) x_i x_i^T \right] u_\ell \\
  &= \sum_{i=1}^N \sum_{k=1}^K \sum_{\ell=1}^K p_k(x_i) (\delta_{k\ell} - p_\ell(x_i)) (u_k^T x_i)(u_\ell^T x_i).
\end{align*}
Define the real scalar $a_{ik} \equiv u_k^T x_i \in \mathbb{R}$. Expanding the double sum over classes for sample $i$:
\begin{align*}
  \sum_{k=1}^K \sum_{\ell=1}^K p_k(x_i)(\delta_{k\ell} - p_\ell(x_i)) a_{ik} a_{i\ell} &= \sum_{k=1}^K p_k(x_i) a_{ik}^2 - \sum_{k=1}^K p_k(x_i) a_{ik} \sum_{\ell=1}^K p_\ell(x_i) a_{i\ell} \\
  &= \sum_{k=1}^K p_k(x_i) a_{ik}^2 - \left( \sum_{k=1}^K p_k(x_i) a_{ik} \right)^2.
\end{align*}
Notice that since $p_k(x_i) \ge 0$ and $\sum_{k=1}^K p_k(x_i) = 1$, the vector $p(x_i)$ defines a valid discrete probability distribution. Therefore, this expression is identically the variance of a discrete random variable $A_i$ taking value $a_{ik}$ with probability $p_k(x_i)$:
\[
  \sum_{k=1}^K p_k(x_i) a_{ik}^2 - \left( \sum_{k=1}^K p_k(x_i) a_{ik} \right)^2 = \operatorname{Var}_{Z \sim p(x_i)}(A_i) \ge 0.
\]
Because the variance of any real-valued variable is non-negative:
\[
  u^T (-\mathcal{H}) u = \sum_{i=1}^N \operatorname{Var}_{Z \sim p(x_i)}(A_i) \ge 0 \quad \forall u \in \mathbb{R}^{K(p+1)}.
\]
Thus, $-\mathcal{H} \succeq 0$, which means that $\mathcal{H} = \nabla^2 \ell(\theta) \preceq 0$ everywhere in parameter space.
\textbf{Conclusion:} The log-likelihood function $\ell(\theta)$ is \textbf{globally concave}. This guarantees that the optimization landscape is free of sub-optimal local maxima, ensuring that standard first-order (gradient ascent) and second-order (Newton-Raphson / IRLS) algorithms are guaranteed to converge to the unique global maximum likelihood estimate. $\blacksquare$

\problemheader{Problem 4: Feature Selection \& Data Leakage in Cross-Validation}{[25 points]}

\textbf{Problem Statement:}
Consider a synthetic dataset of $N = 50$ samples and $p = 5000$ independent standard normal predictors $X_{ij} \sim \mathcal{N}(0, 1)$, and binary targets $Y_i \sim \text{Bernoulli}(0.5)$ sampled independently. The true generalization Bayes error rate is $\operatorname{Err}_{\text{true}} = 50\%$.
\begin{enumerate}[label=(\alph*), itemsep=3pt, leftmargin=18pt]
  \item Explain why selecting the top $100$ features on the entire $N=50$ dataset prior to 10-fold cross-validation reports an error rate near $0\%$ (data leakage).
  \item Derive the expected maximum correlation $\mathbb{E}[\max_j |r_j|]$ across $5000$ noise features using extreme value theory.
  \item State the correct protocol for cross-validation with supervised feature selection to guarantee zero leakage.
\end{enumerate}

\vspace{8pt}
\noindent{\large\bfseries\textcolor{utblue}{Solution:}}
\vspace{4pt}

\textbf{Part (a): Mechanism of the Flawed Protocol (Data Leakage) [9 points]} \\
In the flawed validation protocol, the practitioner computes the sample correlation $r_j = \widehat{\operatorname{Corr}}(X_j, Y)$ across all $N = 50$ samples of the dataset \textbf{prior} to performing cross-validation splits.
\begin{enumerate}[label=\arabic*., leftmargin=18pt, itemsep=3pt]
  \item \textbf{Selection Bias on the Entire Sample:} The $m = 100$ features in $\mathcal{S}$ are selected specifically because their sample correlation with the complete target vector $Y = [Y_1, \dots, Y_{50}]^T$ is abnormally high.
  \item \textbf{Leakage into the Validation Holdout:} When 10-fold cross-validation is performed, each fold holds out 5 test samples. However, those 5 test samples have \textbf{already contributed} to selecting $\mathcal{S}$!
  \item \textbf{Artificial Correlation in Test Folds:} Because the features in $\mathcal{S}$ were chosen precisely because their values happened to correlate with the observed $Y$ in those specific 5 holdout observations, the test samples are no longer independent out-of-sample evaluations.
  \item \textbf{Overoptimistic Reported Performance:} A classifier (such as 1-NN or LDA) trained on the 45 remaining samples will find that these 100 features generalize extraordinarily well to the 5 holdout samples purely because the noise was co-selected. The cross-validation error rate drops to near $0\%$, fatally misleading the researcher into believing they have found a highly predictive signature when the true generalization rate on genuine unseen data is exactly $50\%$ (random guessing). $\blacksquare$
\end{enumerate}

\vspace{4pt}
\textbf{Part (b): Extreme Value Statistics of Spurious Correlation [7 points]} \\
Under the null hypothesis of independence ($X_{ij} \perp Y_i$), the empirical sample correlation coefficient $r_j$ computed on $N = 50$ observations has mean zero and standard error:
\[
  \operatorname{SE}(r_j) \approx \frac{1}{\sqrt{N}} = \frac{1}{\sqrt{50}} \approx 0.1414.
\]
The normalized statistic $Z_j = \frac{r_j}{\operatorname{SE}(r_j)} \approx \sqrt{50} \, r_j$ is asymptotically distributed as standard normal $\mathcal{N}(0, 1)$.
Because all $p = 5000$ predictors are generated independently, we are observing $5000$ independent realizations of $Z_j$.

By the classical Fisher-Tippett-Gnedenko Extreme Value Theorem for Gaussian maxima, the expectation of the maximum order statistic across $p$ independent standard normals is:
\[
  \mathbb{E}\left[ \max_{1 \le j \le p} Z_j \right] \approx \sqrt{2 \ln p} - \frac{\ln(\ln p) + \ln(4\pi)}{2\sqrt{2\ln p}} \approx \sqrt{2 \ln 5000} \approx \sqrt{2 \times 8.517} = \sqrt{17.034} \approx 4.127.
\]
Translating back to the scale of sample correlations:
\[
  \mathbb{E}\left[ \max_{1 \le j \le 5000} |r_j| \right] \approx 4.127 \times \frac{1}{\sqrt{50}} \approx 4.127 \times 0.1414 \approx 0.584.
\]
Furthermore, consider the probability that an individual noise feature achieves an absolute sample correlation exceeding $|r_j| \ge 0.35$ (corresponding to $|Z_j| \ge 0.35 \times \sqrt{50} \approx 2.475$):
\[
  P(|r_j| \ge 0.35) = 2 \left( 1 - \Phi(2.475) \right) \approx 2 \times 0.00666 = 0.0133.
\]
Across $p = 5000$ independent noise features, the expected number of spurious features with $|r_j| \ge 0.35$ is:
\[
  \mathbb{E}\left[\#\{j : |r_j| \ge 0.35\}\right] = 5000 \times 0.0133 \approx 66.6 \text{ features.}
\]
Thus, finding 100 features with sample correlations between $0.35$ and $0.60$ is virtually guaranteed purely by chance. When $p \gg N$, extreme value statistics guarantees that spurious correlations will mimic real biological or physical signals if feature selection precedes cross-validation splitting. $\blacksquare$

\vspace{4pt}
\textbf{Part (c): The Correct Cross-Validation Protocol [9 points]} \\
To ensure completely unbiased model validation, \textbf{all supervised preprocessing and feature screening must be embedded strictly inside each cross-validation fold}:
\begin{enumerate}[label=\arabic*., leftmargin=18pt, itemsep=3pt]
  \item \textbf{Partition the Data First:} Partition the complete dataset of $N=50$ observations into $K=10$ disjoint folds $F_1, \dots, F_{10}$ of 5 samples each.
  \item \textbf{Iterate Across Folds ($k = 1, \dots, 10$):}
  \begin{itemize}[itemsep=2pt, leftmargin=14pt]
    \item \textbf{Split:} Set aside fold $F_k$ as the holdout validation set $\mathcal{D}_{\text{val}}^{(k)}$ ($5$ samples). The remaining $9$ folds constitute the training set $\mathcal{D}_{\text{train}}^{(-k)} = \mathcal{D} \setminus F_k$ ($45$ samples).
    \item \textbf{Screen Features on Training Only:} Compute sample correlations $r_j^{(-k)}$ between features and labels using \textbf{only} the 45 observations in $\mathcal{D}_{\text{train}}^{(-k)}$.
    \item \textbf{Select Features:} Determine the set $\mathcal{S}_k \subset \{1, \dots, 5000\}$ of $100$ features having the highest $|r_j^{(-k)}|$.
    \item \textbf{Train Classifier:} Fit the classifier (e.g., 1-NN or LDA) using only the subset of features $\mathcal{S}_k$ on $\mathcal{D}_{\text{train}}^{(-k)}$.
    \item \textbf{Validate on Unseen Holdout:} Predict the class labels for the 5 holdout observations in $\mathcal{D}_{\text{val}}^{(k)}$ using the model trained on $\mathcal{S}_k$. Record the validation error rate $\operatorname{Err}_k$.
  \end{itemize}
  \item \textbf{Aggregate Performance:} Compute the overall cross-validation estimate:
  \[
    \operatorname{CV}_{(10)} = \frac{1}{10} \sum_{k=1}^{10} \operatorname{Err}_k.
  \]
\end{enumerate}
\textbf{Theoretical Guarantee:} Because the holdout fold $F_k$ was completely quarantined while selecting $\mathcal{S}_k$, the features in $\mathcal{S}_k$ have zero expected correlation with $Y$ on $F_k$. When this proper protocol is executed on pure noise, $\operatorname{CV}_{(10)}$ correctly yields an error rate of approximately $50\%$, accurately exposing the lack of true predictive signal. $\blacksquare$
